Zalo Bot & Thông báo

Kết nối Bot

Lấy Bot Token & Chat ID nhóm từ Zalo Bot Creator (zalo.me/s/botcreator)
Sau khi triển khai xong (chỉ 1 lần, xem hướng dẫn ngay bên dưới), bạn chỉ cần nhập đúng 2 ô ở trên: Bot Token và Chat ID nhóm rồi bấm "Lưu cấu hình" — ô "Máy chủ trung gian" đã được điền sẵn đúng URL khớp với project Firebase hiện tại (${firebaseConfig.projectId}), không cần sửa. Mọi việc còn lại (gửi lịch mỗi sáng, lịch trực tuần sau, nhắc lịch, thông báo định kỳ) sẽ tự chạy nền 24/7 trên máy chủ của Google — kể cả khi không ai mở trang web này.

Vì sao cần "máy chủ trung gian" thay vì gọi thẳng từ trình duyệt? API Zalo Bot (bot-api.zapps.me) chỉ cho phép gọi server-to-server bằng Bot Token bí mật (giống Telegram Bot API), trình duyệt sẽ luôn bị chặn CORS — đây không phải lỗi của trang này. Đồng thời, nếu chỉ chạy bằng JavaScript trong trình duyệt thì bot sẽ ngừng hoạt động ngay khi bạn tắt trang/tắt máy. Vì vậy phần "chạy nền 24/7" bắt buộc phải là một máy chủ thật sự do Google vận hành liên tục — đây là giới hạn của mọi trang web tĩnh, không riêng gì trang này, và Claude cũng không thể tự thay bạn "bật" một máy chủ đang chạy hộ được — bạn cần triển khai (deploy) đúng 1 lần theo các bước dưới đây, sau đó nó tự chạy mãi mãi, không cần đụng vào nữa.

Hướng dẫn triển khai máy chủ trung gian (làm 1 lần — khoảng 10 phút)

1. Cài Node.js rồi cài Firebase CLI: npm install -g firebase-tools
2. Đăng nhập: firebase login (dùng đúng tài khoản Google chủ của project ${firebaseConfig.projectId})
3. Tạo một thư mục trống bất kỳ trên máy, mở terminal tại đó, chạy: firebase init functions → chọn "Use an existing project" → chọn ${firebaseConfig.projectId} → chọn JavaScript → không cần ESLint → có, cài đặt dependencies.
4. Mở file functions/index.js vừa tạo, xoá hết nội dung mặc định và dán đè bằng toàn bộ đoạn code ở khung bên dưới.
5. Trong terminal, tại thư mục functions, chạy thêm: npm install node-fetch@2
6. Bấm nâng cấp project lên gói Blaze (trả theo dùng) tại Firebase Console nếu chưa nâng — bắt buộc để dùng lịch chạy nền (pubsub schedule) và gọi mạng ra ngoài; với quy mô 1 lớp học, chi phí thực tế gần như luôn là 0đ vì nằm trong hạn mức miễn phí hàng tháng.
7. Triển khai: firebase deploy --only functions
8. Xong! Quay lại đây, nhập Bot Token và Chat ID nhóm, bấm "Lưu cấu hình" rồi "Gửi thử ngay" để kiểm tra.

const functions = require('firebase-functions'); const admin = require('firebase-admin'); const fetch = require('node-fetch'); admin.initializeApp(); const db = admin.database(); const WEEKDAYS = ['Thứ 2','Thứ 3','Thứ 4','Thứ 5','Thứ 6','Thứ 7','Chủ nhật']; function pad(n){ return n<10 ? '0'+n : ''+n; } function nowVN(){ // Giờ hiện tại theo múi giờ Việt Nam, không phụ thuộc múi giờ máy chủ return new Date(new Date().toLocaleString('en-US', { timeZone: 'Asia/Ho_Chi_Minh' })); } function dateKey(d){ return d.getFullYear()+'-'+pad(d.getMonth()+1)+'-'+pad(d.getDate()); } function addDays(d,n){ const r=new Date(d); r.setDate(r.getDate()+n); return r; } function fmtFull(d){ return pad(d.getDate())+'/'+pad(d.getMonth()+1)+'/'+d.getFullYear(); } function fmtShort(d){ return pad(d.getDate())+'/'+pad(d.getMonth()+1); } function dowOf(d){ return (d.getDay()+6)%7; } // 0=Thứ2 .. 6=CN function getMonday(d){ const dow=dowOf(d); return addDays(d,-dow); } async function loadAll(){ const [students, items, duty, taskTypes, classCfg, zaloCfg, notifs, leaves] = await Promise.all([ db.ref('students').once('value').then(s=>s.val()||{}), db.ref('scheduleItems').once('value').then(s=>s.val()||{}), db.ref('duty').once('value').then(s=>s.val()||{}), db.ref('dutyTaskTypes').once('value').then(s=>s.val()||{}), db.ref('classConfig').once('value').then(s=>s.val()||{}), db.ref('zaloConfig').once('value').then(s=>s.val()||{}), db.ref('zaloNotifications').once('value').then(s=>s.val()||{}), db.ref('leaveRequests').once('value').then(s=>s.val()||{}) ]); return { students, items, duty, taskTypes, classCfg, zaloCfg, notifs, leaves }; } function itemsForDate(items, d){ const key = dateKey(d), dow = dowOf(d); return Object.values(items||{}).filter(it=>{ if(it.recurrence==='once') return it.onDate===key; return it.dayOfWeek===dow; }).sort((a,b)=> (a.startTime||'').localeCompare(b.startTime||'')); } function absentListForDate(leaves, dk){ const out = []; Object.values(leaves||{}).forEach(r=>{ if(r.type!=='off' || r.status==='rejected') return; if(dk>=r.dateFrom && dk<=r.dateTo) out.push({ studentId:r.studentId, name:r.studentName, status:r.status }); }); return out; } function buildDailyMessage(data, today){ const dk = dateKey(today), dow = dowOf(today); const items = itemsForDate(data.items, today); const lessons = items.filter(i=>i.type==='lesson'); const events = items.filter(i=>i.type==='event'); const dayDuty = (data.duty||{})[dk] || {}; let msg = `LỊCH NGÀY ${fmtFull(today)} (${WEEKDAYS[dow]})\n`; msg += `Lớp ${data.classCfg.className||'Địa 2'} · ${data.classCfg.khoa||'2026-2029'}\n\n`; msg += `Lịch học:\n`; msg += lessons.length? lessons.map(l=>`• ${l.startTime}${l.endTime?'-'+l.endTime:''} ${l.subject}${l.teacher?' ('+l.teacher+')':''}`).join('\n') : '• Không có tiết học nào được ghi nhận.'; msg += `\n\nSự kiện:\n`; msg += events.length? events.map(e=>`• ${e.startTime}${e.endTime?'-'+e.endTime:''} ${e.title}${e.content?' — '+e.content:''}`).join('\n') : '• Không có sự kiện.'; msg += `\n\nLịch trực nhật:\n`; const dutyLines = Object.entries(dayDuty).filter(([tid,arr])=>(arr||[]).length>0).map(([tid,arr])=>{ const t = data.taskTypes[tid]; const names = arr.map(sid=> data.students[sid]? data.students[sid].name : sid).join(', '); return `• ${t?t.name:tid}: ${names}`; }); msg += dutyLines.length? dutyLines.join('\n') : '• Chưa phân công trực nhật cho hôm nay.'; if(data.zaloCfg.headcountEnabled!==false){ const total = Object.keys(data.students||{}).length; const absent = absentListForDate(data.leaves, dk); msg += `\n\nSĩ số dự kiến hôm nay: ${total-absent.length}/${total}`; if(absent.length) msg += `\n• Vắng: ${absent.map(a=>a.name+(a.status==='approved'?' (Đã duyệt)':'')).join(', ')}`; } return msg; } function buildNextWeekDutyMessage(data, today){ const nextMonday = addDays(getMonday(today),7); const days = [...Array(6)].map((_,i)=> addDays(nextMonday,i)); let msg = `LỊCH TRỰC NHẬT TUẦN TỚI (${fmtShort(nextMonday)} – ${fmtFull(addDays(nextMonday,5))})\n`; msg += `Lớp ${data.classCfg.className||'Địa 2'} · ${data.classCfg.khoa||'2026-2029'}\n\n`; days.forEach(d=>{ const dk = dateKey(d), dow = dowOf(d); const dayDuty = (data.duty||{})[dk] || {}; const lines = Object.entries(dayDuty).filter(([tid,arr])=>(arr||[]).length>0).map(([tid,arr])=>{ const t = data.taskTypes[tid]; const names = arr.map(sid=> data.students[sid]? data.students[sid].name : sid).join(', '); return ` • ${t?t.name:tid}: ${names}`; }); msg += `${WEEKDAYS[dow]} (${fmtShort(d)}):\n` + (lines.length? lines.join('\n') : ' • Chưa phân công') + '\n'; }); return msg; } function mergeTemplate(tpl, data, today){ const dk = dateKey(today); const items = itemsForDate(data.items, today); const lessons = items.filter(i=>i.type==='lesson').map(l=>`${l.startTime} ${l.subject}`).join(', ') || 'không có'; const events = items.filter(i=>i.type==='event').map(e=>`${e.startTime} ${e.title}`).join(', ') || 'không có'; const dayDuty = (data.duty||{})[dk] || {}; const truc = Object.entries(dayDuty).filter(([tid,arr])=>(arr||[]).length>0).map(([tid,arr])=>{ const t = data.taskTypes[tid]; return `${t?t.name:tid}: ${arr.map(sid=>data.students[sid]?data.students[sid].name:sid).join(', ')}`; }).join(' · ') || 'chưa phân công'; return (tpl||'').replace(/\{lich_hoc\}/g,lessons).replace(/\{su_kien\}/g,events) .replace(/\{truc_nhat\}/g,truc).replace(/\{ngay\}/g,fmtFull(today)).replace(/\{lop\}/g,(data.classCfg.className||'Địa 2')); } async function sendZalo(token, chatId, text){ const r = await fetch(`https://bot-api.zapps.me/bot${token}/sendMessage`, { method:'POST', headers:{'Content-Type':'application/json'}, body: JSON.stringify({ chat_id: chatId, text }) }); const detail = await r.text().catch(()=> ''); await db.ref('zaloLog').push({ ts: Date.now(), status: r.ok?'success':'failed', preview: text.slice(0,120), fullText: text, detail: `HTTP ${r.status} (chạy nền tự động)${detail?' — '+detail.slice(0,200):''}` }); return r.ok; } // 1) "Máy chủ trung gian" — chỉ dùng cho nút "Gửi thử ngay" trên trang (gọi trực tiếp từ trình duyệt, tránh lỗi CORS) exports.zaloSendProxy = functions.https.onRequest(async (req, res) => { res.set('Access-Control-Allow-Origin', '*'); res.set('Access-Control-Allow-Methods', 'POST'); res.set('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { res.status(204).send(''); return; } const { token, chatId, text } = req.body || {}; try { const ok = await sendZalo(token, chatId, text); res.status(ok ? 200 : 502).json({ ok }); } catch (e) { res.status(500).json({ ok:false, error: e.message }); } }); // 2) Chạy nền 24/7 — cứ mỗi phút Google tự gọi hàm này, không phụ thuộc trang web có mở hay không exports.dailyZaloReminder = functions.pubsub .schedule('every 1 minutes').timeZone('Asia/Ho_Chi_Minh') .onRun(async () => { const data = await loadAll(); const c = data.zaloCfg || {}; if(!c.token || !c.chatId) return null; // chưa cấu hình Bot Token / Chat ID thì bỏ qua const now = nowVN(); const hm = pad(now.getHours())+':'+pad(now.getMinutes()); const dk = dateKey(now); const dow = dowOf(now); // Gửi lịch mỗi sáng (Thứ 2 - Thứ 6) if(c.dailyEnabled!==false && hm===(c.dailyTime||'06:30') && dow<=4 && c._lastDailySent!==dk){ await sendZalo(c.token, c.chatId, buildDailyMessage(data, now)); await db.ref('zaloConfig/_lastDailySent').set(dk); } // Gửi lịch trực tuần tới (theo ngày/giờ đã cấu hình, mặc định Chủ nhật 19:00) if(c.weeklyEnabled!==false && hm===(c.weeklyTime||'19:00') && dow===(c.weeklyDay!=null?c.weeklyDay:6) && c._lastWeeklySent!==dk){ await sendZalo(c.token, c.chatId, buildNextWeekDutyMessage(data, now)); await db.ref('zaloConfig/_lastWeeklySent').set(dk); } // Nhắc lịch trước giờ diễn ra — chỉ lịch/sự kiện được tick "reminderEnabled" const minutesAhead = c.reminderMinutes || 30; const todays = itemsForDate(data.items, now).filter(it=> it.reminderEnabled && it.startTime); for(const it of todays){ const [hh,mm] = it.startTime.split(':').map(Number); const start = new Date(now); start.setHours(hh,mm,0,0); const diffMin = (start - now) / 60000; const sentKey = 'reminders/'+dk+'/'+it.id; if(diffMin>=0 && diffMin<=minutesAhead){ const already = (await db.ref('zaloRunLog/'+sentKey).once('value')).val(); if(!already){ const kind = it.type==='lesson' ? 'Tiết học' : 'Sự kiện'; const label = it.type==='lesson' ? it.subject : it.title; await sendZalo(c.token, c.chatId, `NHẮC LỊCH: ${kind} "${label}" sắp diễn ra lúc ${it.startTime}${it.teacher?' — GV: '+it.teacher:''}${it.content?'\n'+it.content:''}`); await db.ref('zaloRunLog/'+sentKey).set(true); } } } // Thông báo định kỳ tuỳ chỉnh for(const [nid, n] of Object.entries(data.notifs||{})){ if(n.active===false || n.time!==hm) continue; const matches = n.schedule==='daily' ? true : n.schedule==='weekly' ? n.dayOfWeek===dow : n.schedule==='once' ? n.date===dk : false; if(!matches) continue; const sentKey = 'notifs/'+dk+'/'+nid; const already = (await db.ref('zaloRunLog/'+sentKey).once('value')).val(); if(already) continue; await sendZalo(c.token, c.chatId, mergeTemplate(n.message, data, now)); await db.ref('zaloRunLog/'+sentKey).set(true); } return null; });

Máy chủ TURN cho Meet tích hợp (khuyến nghị)
Hệ thống dùng STUN của Google để kết nối trực tiếp giữa các máy. Một số mạng (4G/5G, mạng trường học) chặn kết nối trực tiếp — khi đó cần máy chủ TURN để chuyển tiếp. Có thể đăng ký miễn phí tại metered.ca (Open Relay) hoặc dùng coturn tự dựng, rồi dán thông tin vào đây.

Tối ưu tốc độ Firebase (nên làm 1 lần)

Vào Firebase Console → Realtime Database → Rules, thêm chỉ mục sau (giữ nguyên các quy tắc hiện có của bạn). Việc này giúp lịch sử điểm danh tải nhanh dù dữ liệu ngày càng nhiều.

Cài đặt FaceID
FaceID dùng chung cho Kiosk cửa lớp và xác thực lớp học online. Hãy liên kết mỗi khuôn mặt với đúng học sinh để trạng thái điểm danh được đồng bộ.
Secret QR Code
Mã QR bí mật dùng để mở cửa cho khách / sự kiện. Lượt quét QR được ghi vào lịch sử nhưng không tính là điểm danh học sinh vì mã không gắn với một người cụ thể.
